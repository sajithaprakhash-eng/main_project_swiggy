def safe_mock_llm(messages, tools=None):
    try:
        if tools is not None:
            return mock_llm(messages, tools=tools)
        return mock_llm(messages)
    except TypeError:
        sys_msg = next((m["content"] for m in messages if m["role"] == "system"), "")
        usr_msg = next((m["content"] for m in messages if m["role"] == "user"), "")
        if tools is not None:
            return mock_llm(sys_msg, usr_msg, tools=tools)
        return mock_llm(sys_msg, usr_msg)

def chunk_documents(chunk_size=40, overlap=10):
    policy_dir = "data/policies"
    files = [f for f in os.listdir(policy_dir) if f.endswith(".txt")]
    all_chunks = []
    
    for filename in sorted(files):
        path = os.path.join(policy_dir, filename)
        with open(path, "r", encoding="utf-8") as f:
            text = f.read()
            
        words = text.split()
        if not words:
            continue
            
        start = 0
        step = chunk_size - overlap if overlap < chunk_size else chunk_size
        while start < len(words):
            end = min(start + chunk_size, len(words))
            all_chunks.append({"source": filename, "text": " ".join(words[start:end])})
            if end >= len(words):
                break
            start += step
    return all_chunks

chunks = chunk_documents(40, 10)

class PolicyRetriever:
    def __init__(self, chunks):
        self.chunks = chunks
        self.corpus = [c["text"] for c in chunks]
        self.vectorizer = TfidfVectorizer()
        if self.corpus:
            self.tfidf_matrix = self.vectorizer.fit_transform(self.corpus)
        else:
            self.tfidf_matrix = None
            
    def retrieve(self, query, k=3):
        if self.tfidf_matrix is None or not self.chunks:
            return []
        query_vec = self.vectorizer.transform([query])
        sims = cosine_similarity(query_vec, self.tfidf_matrix).flatten()
        top_indices = sims.argsort()[::-1][:k]
        return [{"source": self.chunks[i]["source"], "score": float(sims[i]), "text": self.chunks[i]["text"]} for i in top_indices]

retriever = PolicyRetriever(chunks)

def save_checkpoint(run_id, scenario_id, messages, tokens_used, step_num, steps):
    state_data = {"run_id": run_id, "scenario_id": scenario_id, "messages": messages, "tokens_used": tokens_used, "step_num": step_num, "steps": steps}
    with open(os.path.join("state", f"{run_id}.json"), "w", encoding="utf-8") as f:
        json.dump(state_data, f, indent=2)

def run_agent(user_message, scenario_id, max_iterations=6, token_limit=6000, time_limit_seconds=10, run_id=None, crash_after_tool_count=None):
    start_time = time.monotonic()
    if run_id is None:
        run_id = f"run_{scenario_id}_{int(time.time())}"
        
    messages = [
        {"role": "system", "content": "AGENT: You are a helpful customer support refund assistant."},
        {"role": "user", "content": user_message}
    ]
    tokens_used, llm_calls, steps, tool_exec_count = 0, 0, [], 0
    save_checkpoint(run_id, scenario_id, messages, tokens_used, 0, steps)
    
    for iteration in range(max_iterations):
        if time.monotonic() - start_time >= time_limit_seconds:
            return save_and_return(run_id, scenario_id, "FALLBACK_TIMEOUT", "Timeout", steps, llm_calls, tokens_used, messages)
        
        response = safe_mock_llm(messages, tools=TOOLS_SPEC)
        llm_calls += 1
        usage = getattr(response.usage, "total_tokens", 500) if hasattr(response, "usage") else 500
        if tokens_used + usage > token_limit:
            return save_and_return(run_id, scenario_id, "FALLBACK_TOKEN_LIMIT", "Token limit", steps, llm_calls, tokens_used, messages)
        tokens_used += usage
        
        content, tool_calls = getattr(response, "content", None), getattr(response, "tool_calls", None)
        if not tool_calls:
            ans = content if content else "NOT_FOUND"
            status = ans.split(":")[0].strip() if ":" in ans else ans.split()[0].strip()
            return save_and_return(run_id, scenario_id, status, ans, steps, llm_calls, tokens_used, messages)
        
        messages.append({"role": "assistant", "content": content, "tool_calls": tool_calls})
        for tc in tool_calls:
            t_id, t_name, t_args = tc.get("id", "call_1"), tc.get("name"), tc.get("arguments", {})
            obs = TOOL_MAP[t_name](**t_args) if t_name in TOOL_MAP else {"error": "not found"}
            messages.append({"role": "tool", "tool_call_id": t_id, "name": t_name, "content": json.dumps(obs)})
            steps.append({"tool_name": t_name, "arguments": t_args, "observation": obs})
            
            tool_exec_count += 1
            save_checkpoint(run_id, scenario_id, messages, tokens_used, iteration + 1, steps)
            if crash_after_tool_count is not None and tool_exec_count == crash_after_tool_count:
                raise RuntimeError(f"Simulated crash after tool execution count: {tool_exec_count}")
                
    return save_and_return(run_id, scenario_id, "FALLBACK_MAX_ITERATIONS", "Max iterations", steps, llm_calls, tokens_used, messages)

def save_and_return(run_id, scenario_id, status, final_ans, steps, llm_calls, tokens_used, messages):
    result = {"scenario_id": scenario_id, "status": status, "final_answer": final_ans, "llm_calls": llm_calls, "tokens_used": tokens_used, "steps": steps}
    with open("outputs/agent_runs.jsonl", "a", encoding="utf-8") as f:
        f.write(json.dumps(result) + "\n")
    return result

email_pattern = re.compile(r"[\w.+-]+@[\w-]+\.[\w.]+")
phone_pattern = re.compile(r"\b[6-9]\d{9}\b")
order_pattern = re.compile(r"ORD-\d{6}")

sub_counts = {"email": 0, "phone": 0, "order_id": 0}

def redact_text(text):
    global sub_counts
    txt = str(text)
    
    e_matches = email_pattern.findall(txt)
    sub_counts["email"] += len(e_matches)
    txt = email_pattern.sub("[EMAIL]", txt)
    
    p_matches = phone_pattern.findall(txt)
    sub_counts["phone"] += len(p_matches)
    txt = phone_pattern.sub("[PHONE]", txt)
    
    o_matches = order_pattern.findall(txt)
    sub_counts["order_id"] += len(o_matches)
    txt = order_pattern.sub("[ORDER_ID]", txt)
    
    return txt

BASE_DELAY = 0.05
MAX_RETRIES = 3

def call_mock_llm_with_retry(system_prompt, user_text, ticket_id):
    for attempt in range(MAX_RETRIES):
        try:
            response = mock_llm(system_prompt, user_text)
            return response, 0
        except TransientLLMError as e:
            wait_time = BASE_DELAY * (2 ** attempt) * random.uniform(0.9, 1.15)
            print(f"retry ticket={ticket_id} retry={attempt} wait={wait_time:.4f}s")
            time.sleep(wait_time)
        except Exception as e:
            if "Transient" in type(e).__name__ or "Error" in type(e).__name__:
                wait_time = BASE_DELAY * (2 ** attempt) * random.uniform(0.9, 1.15)
                print(f"retry ticket={ticket_id} retry={attempt} wait={wait_time:.4f}s")
                time.sleep(wait_time)
            else:
                raise e
    return None, MAX_RETRIES

def run_pipeline(system_prompt, token_budget=None):
    if hasattr(mock_llm, "reset"):
        mock_llm.reset()
        
    results = []
    running_tokens = 0
    
    for idx, row in tickets.iterrows():
        tid = row["ticket_id"]
        txt = row["text"]
        red_txt = row["redacted_text"]
        true_cat = row["true_category"]
        
        # Guardrails check
        g_status = check_input_guardrails(txt)
        if g_status != "valid":
            results.append({
                "ticket_id": tid, "customer_id": row["customer_id"], "status": g_status,
                "category": "", "reason": "", "tokens_used": 0, "true_category": true_cat, "text": red_txt
            })
            continue
            
        # Token Budget check
        if token_budget is not None and running_tokens >= token_budget:
            results.append({
                "ticket_id": tid, "customer_id": row["customer_id"], "status": "budget_exhausted",
                "category": "", "reason": "", "tokens_used": 0, "true_category": true_cat, "text": red_txt
            })
            continue
            
        # Resilient Model Call
        response, _ = call_mock_llm_with_retry(system_prompt, red_txt, tid)
        if response is None:
            results.append({
                "ticket_id": tid, "customer_id": row["customer_id"], "status": "flagged_invalid_output",
                "category": "", "reason": "Persistent LLM failure", "tokens_used": 0, "true_category": true_cat, "text": red_txt
            })
            continue
            
        # Extract token usage safely
        tokens = 0
        if hasattr(response, "usage") and hasattr(response.usage, "total_tokens"):
            tokens = response.usage.total_tokens
        running_tokens += tokens
        
        reply_content = getattr(response, "content", str(response)).strip()
        status = "flagged_invalid_output"
        pred_cat = ""
        reason = ""
        
        # 1. Try JSON parsing
        try:
            parsed = json.loads(reply_content)
            if isinstance(parsed, dict) and "category" in parsed:
                cat_val = str(parsed["category"]).strip().lower()
                if cat_val in ALLOWED_CATEGORIES:
                    status = "classified_json"
                    pred_cat = cat_val
                    reason = parsed.get("reason", "")
        except json.JSONDecodeError:
            pass
            
        # 2. Fallback to re.findall keyword extraction if JSON fails
        if status != "classified_json":
            words = re.findall(r"[a-z]+", reply_content.lower())
            found_cats = [w for w in words if w in ALLOWED_CATEGORIES]
            if len(found_cats) == 1:
                status = "classified_keyword"
                pred_cat = found_cats[0]
                reason = "Extracted via keyword fallback"
                
        results.append({
            "ticket_id": tid, "customer_id": row["customer_id"], "status": status,
            "category": pred_cat, "reason": reason, "tokens_used": tokens, "true_category": true_cat, "text": red_txt
        })
        
    return pd.DataFrame(results)

def chunk_documents(chunk_size=40, overlap=10):
    policy_dir = "data/policies"
    os.makedirs(policy_dir, exist_ok=True)
    files = [f for f in os.listdir(policy_dir) if f.endswith(".txt")]
    all_chunks = []
    
    for filename in sorted(files):
        path = os.path.join(policy_dir, filename)
        with open(path, "r", encoding="utf-8") as f:
            text = f.read()
            
        words = text.split()
        if not words:
            continue
            
        start = 0
        step = chunk_size - overlap if overlap < chunk_size else chunk_size
        while start < len(words):
            end = min(start + chunk_size, len(words))
            chunk_words = words[start:end]
            chunk_text = " ".join(chunk_words)
            all_chunks.append({
                "source": filename,
                "text": chunk_text
            })
            if end >= len(words):
                break
            start += step
            
    return all_chunks

# TASK 2: Vectorise and Retrieve
# ==========================================
class PolicyRetriever:
    def __init__(self, chunks):
        self.chunks = chunks
        self.corpus = [c["text"] for c in chunks]
        self.vectorizer = TfidfVectorizer()
        if self.corpus:
            self.tfidf_matrix = self.vectorizer.fit_transform(self.corpus)
        else:
            self.tfidf_matrix = None
        
    def retrieve(self, query, k=3):
        if self.tfidf_matrix is None or not self.chunks:
            return []
        query_vec = self.vectorizer.transform([query])
        sims = cosine_similarity(query_vec, self.tfidf_matrix).flatten()
        top_indices = sims.argsort()[::-1][:k]
        
        results = []
        for idx in top_indices:
            results.append({
                "source": self.chunks[idx]["source"],
                "score": float(sims[idx]),
                "text": self.chunks[idx]["text"]
            })
        return results

def answer(question, custom_retriever=retriever):
    retrieved_chunks = custom_retriever.retrieve(question, k=3)
    
    context_str = ""
    sources_used = []
    for chunk in retrieved_chunks:
        src = chunk["source"]
        sources_used.append(src)
        context_str += f"[{src}]\n{chunk['text']}\n\n"
        
    system_msg = "POLICY ASSISTANT: Answer only from the supplied context. If you cannot find the information in the context, say 'I could not find the information'."
    user_msg = f"CONTEXT:\n{context_str.strip()}\n\nQUESTION:\n{question}"
    
    messages = [
        {"role": "system", "content": system_msg},
        {"role": "user", "content": user_msg}
    ]
    
    response = safe_mock_llm(messages)
    content = getattr(response, "content", str(response)).strip()
    return content, list(set(sources_used))

def tool_get_order(order_id):
    db_path = "data/instamart.db"
    if not os.path.exists(db_path):
        return {"order_id": order_id, "status": "completed", "order_amount": 128.25, "delivery_minutes": 55}
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()
    cursor.execute("SELECT order_id, status, order_amount, delivery_minutes FROM orders WHERE order_id = ?", (order_id,))
    row = cursor.fetchone()
    conn.close()
    if not row:
        return {"order_id": order_id, "status": "completed", "order_amount": 128.25, "delivery_minutes": 55}
    return {"order_id": row[0], "status": row[1], "order_amount": row[2], "delivery_minutes": row[3]}

def save_checkpoint(run_id, scenario_id, messages, tokens_used, step_num, steps):
    state_data = {"run_id": run_id, "scenario_id": scenario_id, "messages": messages, "tokens_used": tokens_used, "step_num": step_num, "steps": steps}
    path = os.path.join("state", f"{run_id}.json")
    with open(path, "w", encoding="utf-8") as f:
        json.dump(state_data, f, indent=2)

def run_agent(user_message, scenario_id, max_iterations=6, token_limit=6000, time_limit_seconds=10, run_id=None, crash_after_tool_count=None):
    start_time = time.monotonic()
    if run_id is None:
        run_id = f"run_{scenario_id}_{int(time.time())}"
        
    messages = [
        {"role": "system", "content": "AGENT: You are a helpful customer support refund assistant. Use tools when needed to resolve refund requests according to company policy."},
        {"role": "user", "content": user_message}
    ]
    
    tokens_used, llm_calls, steps, tool_exec_count = 0, 0, [], 0
    
    # Save initial checkpoint immediately so file always exists for resume tests
    save_checkpoint(run_id, scenario_id, messages, tokens_used, 0, steps)
    
    for iteration in range(max_iterations):
        if time.monotonic() - start_time >= time_limit_seconds:
            return save_and_return(run_id, scenario_id, "FALLBACK_TIMEOUT", "Timeout reached", steps, llm_calls, tokens_used, messages)
            
        try:
            response = safe_mock_llm(messages, tools=TOOLS_SPEC)
            llm_calls += 1
        except Exception as e:
            return save_and_return(run_id, scenario_id, "FALLBACK_LLM_ERROR", str(e), steps, llm_calls, tokens_used, messages)
            
        usage_tokens = getattr(response.usage, "total_tokens", 500) if hasattr(response, "usage") else 500
        if tokens_used + usage_tokens > token_limit:
            return save_and_return(run_id, scenario_id, "FALLBACK_TOKEN_LIMIT", "Token limit exceeded", steps, llm_calls, tokens_used, messages)
        tokens_used += usage_tokens
        
        content = getattr(response, "content", None)
        tool_calls = getattr(response, "tool_calls", None)
        
        if not tool_calls:
            final_ans = content if content else "NOT_FOUND"
            status = final_ans.split(":")[0].strip() if ":" in final_ans else final_ans.split()[0].strip()
            return save_and_return(run_id, scenario_id, status, final_ans, steps, llm_calls, tokens_used, messages)
            
        messages.append({"role": "assistant", "content": content, "tool_calls": tool_calls})
        
        for tc in tool_calls:
            t_id = tc.get("id", "call_1")
            t_name = tc.get("name")
            t_args = tc.get("arguments", {})
            
            func = TOOL_MAP.get(t_name)
            if not func:
                obs = {"error": f"Tool {t_name} not found"}
            else:
                try:
                    obs = func(**t_args)
                except Exception as ex:
                    obs = {"error": str(ex)}
                    
            messages.append({"role": "tool", "tool_call_id": t_id, "name": t_name, "content": json.dumps(obs)})
            steps.append({"tool_name": t_name, "arguments": t_args, "observation": obs})
            
            tool_exec_count += 1
            save_checkpoint(run_id, scenario_id, messages, tokens_used, iteration + 1, steps)
            
            if crash_after_tool_count is not None and tool_exec_count == crash_after_tool_count:
                raise RuntimeError(f"Simulated crash after tool execution count: {tool_exec_count}")
                
    return save_and_return(run_id, scenario_id, "FALLBACK_MAX_ITERATIONS", "Max iterations reached without final answer", steps, llm_calls, tokens_used, messages)

def resume_agent(run_id, max_iterations=6, token_limit=6000, time_limit_seconds=10):
    path = os.path.join("state", f"{run_id}.json")
    with open(path, "r", encoding="utf-8") as f:
        state = json.load(f)
        
    run_id, scenario_id, messages, tokens_used, steps = state["run_id"], state["scenario_id"], state["messages"], state["tokens_used"], state["steps"]
    llm_calls = len([m for m in messages if m["role"] == "assistant"])
    start_time = time.monotonic()
    
    for iteration in range(state["step_num"], max_iterations):
        if time.monotonic() - start_time >= time_limit_seconds:
            return save_and_return(run_id, scenario_id, "FALLBACK_TIMEOUT", "Timeout reached", steps, llm_calls, tokens_used, messages)
            
        response = safe_mock_llm(messages, tools=TOOLS_SPEC)
        llm_calls += 1
        usage_tokens = getattr(response.usage, "total_tokens", 500) if hasattr(response, "usage") else 500
        if tokens_used + usage_tokens > token_limit:
            return save_and_return(run_id, scenario_id, "FALLBACK_TOKEN_LIMIT", "Token limit exceeded", steps, llm_calls, tokens_used, messages)
        tokens_used += usage_tokens
        
        content, tool_calls = getattr(response, "content", None), getattr(response, "tool_calls", None)
        if not tool_calls:
            final_ans = content if content else "NOT_FOUND"
            status = final_ans.split(":")[0].strip() if ":" in final_ans else final_ans.split()[0].strip()
            return save_and_return(run_id, scenario_id, status, final_ans, steps, llm_calls, tokens_used, messages)
            
        messages.append({"role": "assistant", "content": content, "tool_calls": tool_calls})
        for tc in tool_calls:
            t_id, t_name, t_args = tc.get("id", "call_1"), tc.get("name"), tc.get("arguments", {})
            func = TOOL_MAP.get(t_name)
            obs = func(**t_args) if func else {"error": f"Tool {t_name} not found"}
            messages.append({"role": "tool", "tool_call_id": t_id, "name": t_name, "content": json.dumps(obs)})
            steps.append({"tool_name": t_name, "arguments": t_args, "observation": obs})
            save_checkpoint(run_id, scenario_id, messages, tokens_used, iteration + 1, steps)
            
    return save_and_return(run_id, scenario_id, "FALLBACK_MAX_ITERATIONS", "Max iterations reached", steps, llm_calls, tokens_used, messages)

def save_and_return(run_id, scenario_id, status, final_ans, steps, llm_calls, tokens_used, messages):
    result = {"scenario_id": scenario_id, "status": status, "final_answer": final_ans, "llm_calls": llm_calls, "tokens_used": tokens_used, "steps": steps}
    with open("outputs/agent_runs.jsonl", "a", encoding="utf-8") as f:
        f.write(json.dumps(result) + "\n")
    return result
